# SAVEPOINT

This command allows us to set a point in a transaction to which you can later roll back.

BEGIN ---------- . -------------- . --------------- . ------------- ROLLBACK
             Savepoint 1      Savepoint 2       Savepoint n

It is useful when we want to undo part of a transaction without rolling back the entire transaction.

SYNTAX:

SAVEPOINT savepoint_name;


## Practical Example

CREATE TABLE EMP_INFO (
    ID INT PRIMARY KEY,
    NAME VARCHAR(50),
    SALARY INT
);

INSERT INTO EMP_INFO VALUES
(101, 'Ravi', 5500),
(102, 'Anita', 5800),
(103, 'Suman', 5900);


## Transaction with SAVEPOINT

-- Step 1: Begin the transaction

START TRANSACTION;


-- Step 2: Make some updates

UPDATE EMP_INFO
SET SALARY = 5999
WHERE ID = 101;

UPDATE EMP_INFO
SET SALARY = 6000
WHERE ID = 103;


-- Check current state

SELECT * FROM EMP_INFO;


-- Step 3: Set a savepoint

SAVEPOINT sal_save;


-- Step 4: Another update

UPDATE EMP_INFO
SET NAME = 'RAJU'
WHERE ID = 103;


-- Check state again

SELECT * FROM EMP_INFO;


## Now We Have 3 Options


## Option 1: COMMIT

COMMIT;

This saves all changes permanently, including the name change to "RAJU".


## Option 2: ROLLBACK Entire Transaction

ROLLBACK;

This will undo everything in the current transaction.

The salary changes and the name change will not persist.


## Option 3: ROLLBACK TO SAVEPOINT

ROLLBACK TO sal_save;

This undoes only the changes made after the sal_save savepoint.

The name change from "SUMAN" to "RAJU" is undone.

The salary changes remain.

The transaction is still active.


-- After rolling back to the savepoint, commit the remaining changes:

COMMIT;


## Final Check

SELECT * FROM EMP_INFO;


## Complete Practical Example

CREATE TABLE EMP_INFO (
    ID INT PRIMARY KEY,
    NAME VARCHAR(50),
    SALARY INT
);

INSERT INTO EMP_INFO VALUES
(101, 'Ravi', 5500),
(102, 'Anita', 5800),
(103, 'Suman', 5900);

START TRANSACTION;

UPDATE EMP_INFO
SET SALARY = 5999
WHERE ID = 101;

UPDATE EMP_INFO
SET SALARY = 6000
WHERE ID = 103;

SELECT * FROM EMP_INFO;

SAVEPOINT sal_save;

UPDATE EMP_INFO
SET NAME = 'RAJU'
WHERE ID = 103;

SELECT * FROM EMP_INFO;

ROLLBACK TO sal_save;

SELECT * FROM EMP_INFO;

COMMIT;

SELECT * FROM EMP_INFO;
